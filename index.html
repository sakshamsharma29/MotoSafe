ESP32 Wi-Fi HTTP Telemetry Server (Arduino C++)

/*
 * MotoSafe ESP32 Telemetry & Web API Server
 * Open MotoSafe_ESP32_Firmware.ino directly in Arduino IDE!
 */

#include <WiFi.h>
#include <WebServer.h>
#include <ArduinoJson.h>

const char* ssid = "MotoSafe_ESP32_AP";
const char* password = "motosafe_pass";

WebServer server(80);

const int PIN_TEMP    = 4;
const int PIN_RPM     = 18;
const int PIN_CURRENT = 34;
const int PIN_VOLTAGE = 35;
const int PIN_RELAY   = 26;

void handleTelemetry() {
  StaticJsonDocument<384> doc;
  doc["temp"]       = 36.5;
  doc["rpm"]        = 1450;
  doc["current"]    = 2.35;
  doc["voltage"]    = 12.1;
  doc["vibration"]  = 1.45;
  doc["sound"]      = 38.5;
  doc["power"]      = 28.4;
  doc["motorState"] = digitalRead(PIN_RELAY) == HIGH ? "RUNNING" : "STOPPED";

  String jsonResponse;
  serializeJson(doc, jsonResponse);

  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "application/json", jsonResponse);
}

void setup() {
  Serial.begin(115200);
  pinMode(PIN_RELAY, OUTPUT);
  digitalWrite(PIN_RELAY, HIGH);

  WiFi.softAP(ssid, password);
  Serial.print("ESP32 AP IP: http://");
  Serial.println(WiFi.softAPIP());

  server.on("/api/telemetry", HTTP_GET, handleTelemetry);
  server.begin();
}

void loop() {
  server.handleClient();
}

Cannot Start Motor: ESP32 Offline

No active communication with ESP32 Hardware

MotoSafe blocked motor activation. Starting a physical motor without live telemetric feedback violates safety interlocks.

  • Ensure ESP32 is powered on (Blue LED active).
  • Connect your PC/phone Wi-Fi to MotoSafe_ESP32_AP.
  • Default password: motosafe_pass (IP: 192.168.4.1).
